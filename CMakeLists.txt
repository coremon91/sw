cmake_minimum_required(VERSION 3.24)
project(SW VERSION 0.1.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
include(CTest)
option(SW_BUILD_DESKTOP "Build the Windows desktop switcher" ON)
option(SW_WITH_AJA "Build the AJA NTV2 adapter" ON)
set(SW_AJA_ROOT "${CMAKE_SOURCE_DIR}/vendor-sdk/aja" CACHE PATH "AJA libajantv2 source")
set(SW_DECKLINK_DLL "C:/Program Files/Blackmagic Design/Blackmagic Desktop Video/DeckLinkAPI64.dll" CACHE FILEPATH "Installed DeckLink type library")
add_library(sw_core native/core.cpp)
target_include_directories(sw_core PUBLIC native)
if(MSVC)
  target_compile_options(sw_core PUBLIC /utf-8 /W4 /permissive-)
endif()
if(BUILD_TESTING)
  add_executable(sw_core_tests tests/core_tests.cpp)
  target_link_libraries(sw_core_tests PRIVATE sw_core)
  add_test(NAME core COMMAND sw_core_tests)
endif()
if(SW_BUILD_DESKTOP)
  if(NOT WIN32 OR NOT MSVC)
    message(FATAL_ERROR "The desktop engine requires Windows x64 and MSVC. Use SW_BUILD_DESKTOP=OFF for portable core tests.")
  endif()
  find_package(Qt6 6.8 REQUIRED COMPONENTS Widgets)
  add_library(sw_io native/io.cpp native/decklink.cpp)
  # #import must run in a single compiler process to emit its type-library header.
  set_source_files_properties(native/decklink.cpp PROPERTIES VS_SETTINGS "MultiProcessorCompilation=false")
  target_link_libraries(sw_io PUBLIC sw_core ole32 oleaut32)
  target_compile_definitions(sw_io PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
  if(EXISTS "${SW_DECKLINK_DLL}")
    configure_file(native/decklink_import.hpp.in generated/decklink_import.hpp @ONLY)
    target_include_directories(sw_io PRIVATE "${CMAKE_BINARY_DIR}/generated")
    target_compile_definitions(sw_io PRIVATE SW_HAS_DECKLINK=1)
  endif()
  if(SW_WITH_AJA)
    if(NOT EXISTS "${SW_AJA_ROOT}/CMakeLists.txt")
      message(FATAL_ERROR "Run scripts/bootstrap.ps1 to obtain the pinned AJA SDK, or configure SW_WITH_AJA=OFF.")
    endif()
    set(AJANTV2_DISABLE_DEMOS ON CACHE BOOL "" FORCE)
    set(AJANTV2_DISABLE_TESTS ON CACHE BOOL "" FORCE)
    set(AJANTV2_DISABLE_TOOLS ON CACHE BOOL "" FORCE)
    set(AJANTV2_DISABLE_DRIVER ON CACHE BOOL "" FORCE)
    set(AJANTV2_DISABLE_PLUGIN_LOAD ON CACHE BOOL "" FORCE)
    set(AJANTV2_DISABLE_RDMA ON CACHE BOOL "" FORCE)
    set(AJANTV2_BUILD_SHARED OFF CACHE BOOL "" FORCE)
    set(AJA_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>DLL" CACHE STRING "" FORCE)
    add_subdirectory("${SW_AJA_ROOT}" "${CMAKE_BINARY_DIR}/aja" EXCLUDE_FROM_ALL)
    target_sources(sw_io PRIVATE native/aja.cpp)
    target_link_libraries(sw_io PRIVATE ajantv2)
    target_compile_definitions(sw_io PRIVATE SW_HAS_AJA=1)
  endif()
  add_library(sw_engine native/gpu.cpp native/engine.cpp)
  target_link_libraries(sw_engine PUBLIC sw_io d3d11 dxgi d3dcompiler winmm)
  target_compile_definitions(sw_engine PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
  add_executable(sw_probe native/probe.cpp)
  target_link_libraries(sw_probe PRIVATE sw_io)
  if(BUILD_TESTING)
    add_executable(sw_io_tests tests/io_tests.cpp)
    target_link_libraries(sw_io_tests PRIVATE sw_io)
    add_test(NAME routing COMMAND sw_io_tests)
  endif()
  add_executable(sw_bench native/bench.cpp)
  target_link_libraries(sw_bench PRIVATE sw_engine)
  add_executable(sw_loopback native/loopback.cpp)
  target_link_libraries(sw_loopback PRIVATE sw_io winmm)
  target_compile_definitions(sw_loopback PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
  add_executable(sw_switcher WIN32 native/main.cpp native/window.cpp)
  target_link_libraries(sw_switcher PRIVATE sw_engine Qt6::Widgets)
  install(TARGETS sw_switcher sw_probe sw_bench sw_loopback RUNTIME DESTINATION .)
endif()
